const webpack = require('webpack'); module.exports = function override(config) { // Provide global polyfills config.plugins = (config.plugins || []).concat([ new webpack.ProvidePlugin({ process: 'process/browser', Buffer: ['buffer', 'Buffer'], util: 'util', }), ]); // Add fallbacks config.resolve = { ...config.resolve, fallback: { util: require.resolve('util/'), buffer: require.resolve('buffer/'), process: require.resolve('process/browser'), stream: require.resolve('stream-browserify'), crypto: require.resolve('crypto-browserify'), path: false, fs: false, net: false, tls: false, }, }; return config; };
